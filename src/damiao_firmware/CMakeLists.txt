cmake_minimum_required(VERSION 3.16)
project(forge_arm C)
set(CMAKE_C_STANDARD 11)
add_library(forge_arm src/arm.c src/wire.c src/tx_wait.c)
target_include_directories(forge_arm PUBLIC include)
target_compile_options(forge_arm PRIVATE -Wall -Wextra -Werror)
include(CTest)
if(BUILD_TESTING)
  add_executable(arm_test ../../tests/damiao_arm_test.c)
  target_link_libraries(arm_test PRIVATE forge_arm m)
  target_compile_options(arm_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME damiao_arm COMMAND arm_test)
  add_executable(tx_wait_test ../../tests/damiao_tx_wait_test.c)
  target_link_libraries(tx_wait_test PRIVATE forge_arm)
  target_compile_options(tx_wait_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME damiao_tx_wait COMMAND tx_wait_test)
endif()
